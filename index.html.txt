<script>
// تهيئة Telegram WebApp
const tg = window.Telegram.WebApp;
if (tg) {
tg.expand();
}
function solve() {
const op = document.getElementById('operation').value;
const expr = document.getElementById('expression').value.trim();
const resultDiv = document.getElementById('result');
const resultContainer = document.getElementById('result-container');
if (!expr) {
alert("يرجى كتابة معادلة أولاً!");
return;
}
try {
let output = "";
if (op === 'derivative') {
// حساب المشتقة باستخدام Algebrite
output = Algebrite.derivative(expr, 'x').toString();
} else if (op === 'integral') {
// حساب التكامل باستخدام Algebrite
output = Algebrite.integral(expr, 'x').toString() + " + C";
}
resultDiv.innerText = output;
resultContainer.style.display = "block";
} catch (error) {
resultDiv.innerText = "خطأ في صيغة المعادلة، يرجى التأكد من كتابتها بشكل صحيح.";
resultContainer.style.display = "block";
}
}
</script>
